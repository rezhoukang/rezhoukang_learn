class Calc {
    // 053 方法重载：同类中，方法名相同，参数个数/类型不同，与返回值无关
    //意思是多重负载，而不是重新加载
    int add(int a, int b) {
        return a + b;
    }
    int add(int a, int b, int c) {
        return a + b + c;
    }

    // 057 递归：方法自己调用自己，必须有终止条件
    // 计算阶乘 n! = 1*2*3...*n
    int factorial(int n) {
        if (n == 1) return 1; // 终止条件
        return n * factorial(n - 1);
    }
}

public class E4_OverLoadRecursion {
    public static void main(String[] args) {
        Calc c = new Calc();
        System.out.println(c.add(1,2)); // 输出：3
        System.out.println(c.add(1,2,3)); // 输出：6
        System.out.println("5的阶乘：" + c.factorial(5)); // 输出：5的阶乘：120
    }
}
