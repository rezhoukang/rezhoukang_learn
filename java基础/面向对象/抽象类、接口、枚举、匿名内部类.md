// ！！！抽象是一个类，类里有很多普通方法和抽象方法；接口就是一个放方法声明的地方，可以放多个方法；抽象类和接口被继承后都必须要实现里面所有的抽象方法！！！！


// 063/064 抽象类abstract：有抽象方法，不能new，必须子类实现全部抽象方法
abstract class Shape {
    // 普通方法（有方法体）— 抽象类可以自己写好一部分
    public void setColor(String color) {
        System.out.println("设置颜色为：" + color);
    }
    public void setSize(int size) {
        System.out.println("设置大小为：" + size);
    }
    // 抽象方法（无方法体）— 留空让子类自己实现
    public abstract void draw();
    public abstract void calcArea();// 不写public也是 public abstract
}

// 065 接口interface：标准规范，全抽象方法(Java8前)，多实现
interface Play {
    void game();// 实际是 public abstract void game();
    void watch();// 实际是 public abstract void watch();
}
interface Eat {
    void eat();
}

// ★ 关键区别：一个类只能继承1个抽象类（单继承），但可以实现多个接口（多实现）
class Circle extends Shape implements Play, Eat {
    @Override
    public void draw() {
        System.out.println("画圆形");
    }
    @Override
    public void calcArea() {
        System.out.println("圆形面积 = πr²");
    }
    @Override
    public void game() {
        System.out.println("玩游戏");
    }
    @Override
    public void watch() {
        System.out.println("看视频");
    }
    @Override
    public void eat() {
        System.out.println("吃东西");
    }
}

// 066 枚举：固定常量集合
enum Season { SPRING, SUMMER, AUTUMN, WINTER }

public class E6_AbstractInterface {
    public static void main(String[] args) {
        // 写法一：用父类 Shape 接收 → 只能调 Shape 里的方法，调子类特有方法需要强转
        Shape s = new Circle();
        s.draw(); // 输出：画圆形
        ((Circle)s).game(); // 输出：玩游戏（强转后才能调 Circle 从接口实现的方法）

        // 写法二：直接用子类 Circle 接收 → 不用强转，所有方法随便调（更省事）
        // Circle s2 = new Circle();
        // s2.draw();
        // s2.game();

        Season now = Season.SPRING;
        System.out.println(now); // 输出：SPRING

        // 067 匿名内部类：不用定义子类，临时重写抽象方法
        Shape temp = new Shape() {
            @Override
            public void draw() {
                System.out.println("匿名内部类绘图");
            }
        };
        temp.draw(); // 输出：匿名内部类绘图
    }
}
