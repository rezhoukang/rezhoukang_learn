```java
import java.util.*;

public class H_CollectionAll {
    public static void main(String[] args) {
        // ===================== 098 集合介绍、099 顶层接口 =====================
        /*
        集合和数组区别：
        1.数组长度固定；集合长度可变
        2.数组可存基本/引用类型；集合只能存引用类型（基本类型自动装箱）
        三大分支：
        Collection单列集合（List/Set/Queue）、Map双列键值对集合
        */

        // ===================== H2 List 有序、可重复、有索引 =====================
        // 100 ArrayList 底层数组，查询快增删慢
        List<String> arrayList = new ArrayList<>();
        // 增
        arrayList.add("张三");
        arrayList.add("李四");
        arrayList.add("张三"); // List允许重复元素
        System.out.println("ArrayList全部元素：" + arrayList); // 输出：[张三, 李四, 张三]

        // 删
        arrayList.remove("李四");
        System.out.println("删除李四后：" + arrayList); // 输出：[张三, 张三]

        // 改
        arrayList.set(0, "小明");
        System.out.println("修改下标0元素：" + arrayList); // 输出：[小明, 张三]

        // 查
        System.out.println("下标1元素：" + arrayList.get(1)); // 输出：张三
        System.out.println("集合长度：" + arrayList.size()); // 输出：2

        // 101 遍历集合三种方式
        System.out.println("=====for循环遍历=====");
        for (int i = 0; i < arrayList.size(); i++) {
            System.out.print(arrayList.get(i) + " "); // 输出：小明 张三
        }
        System.out.println("\n=====增强for遍历=====");
        // 把 arrayList 中的每个元素依次取出来，赋值给 s
        for (String s : arrayList) {
            System.out.print(s + " "); // 输出：小明"空格"张三
        }
        System.out.println("\n=====迭代器遍历=====");//iterator：迭代器
        Iterator<String> it = arrayList.iterator();  // 拿一个"指针"指向集合开头
        while (it.hasNext()) {           // hasNext()：后面还有元素吗？true=有
            System.out.print(it.next() + " "); // next()：指针移向下一个，取出该元素
        }
        System.out.println();

        // 102 LinkedList 底层双向链表，首尾增删效率高
        LinkedList<Integer> linkList = new LinkedList<>();
        linkList.addFirst(1); // 头部添加
        linkList.addLast(99); // 尾部添加
        System.out.println("LinkedList首尾添加：" + linkList); // 输出：[1, 99]
        System.out.println("获取首元素：" + linkList.getFirst()); // 输出：1


        // ===================== H3 Set 无序、不可重复、无索引 =====================
        // 104/105 泛型：限制集合只能存指定类型，避免类型转换异常
        Set<String> hashSet = new HashSet<>();
        hashSet.add("苹果");
        hashSet.add("香蕉");
        hashSet.add("苹果"); // 重复元素自动丢弃
        System.out.println("HashSet自动去重：" + hashSet); // 输出：[苹果, 香蕉] 顺序不固定

        // 111 Queue队列：先进先出
        Queue<String> queue = new LinkedList<>();
        queue.offer("消息1");
        queue.offer("消息2");
        System.out.println("队首元素：" + queue.peek()); // 输出：消息1
        System.out.println("取出队首：" + queue.poll()); // 输出：消息1


        // ===================== H4 Map 双列集合：键key唯一，值value可重复 =====================
        // 112 HashMap 开发最常用键值对容器
        Map<Integer, String> studentMap = new HashMap<>();
        // put存键值，key重复会覆盖旧value
        studentMap.put(1001, "张三");
        studentMap.put(1002, "李四");
        studentMap.put(1001, "小张");
        System.out.println("HashMap存储：" + studentMap); // 输出：{1001=小张, 1002=李四}

        // get根据key取值
        System.out.println("学号1002对应姓名：" + studentMap.get(1002)); // 输出：李四

        // 遍历键、遍历值、遍历键值对
        System.out.println("全部key：" + studentMap.keySet()); // 输出：[1001, 1002]
        System.out.println("全部value：" + studentMap.values()); // 输出：[小张, 李四]


        // ===================== H5 工具类Collections、迭代器 =====================
        ArrayList<Integer> numList = new ArrayList<>();
        numList.add(5);
        numList.add(1);
        numList.add(3);
        Collections.sort(numList); // 工具类排序
        System.out.println("Collections排序后：" + numList); // 输出：[1, 3, 5]
    }
}
```
