public class C_AllOperatorFull {
    public static void main(String[] args) {
        // ===================== 017 运算符总介绍 =====================
        // 运算符：用来对变量、数值做计算、比较、判断的符号
        int a = 10;
        int b = 3;

        // ===================== 018 二元算术运算符 + - * / % =====================
        System.out.println("a + b = " + (a + b)); // 输出：a + b = 13
        System.out.println("a - b = " + (a - b)); // 输出：a - b = 7
        System.out.println("a * b = " + (a * b)); // 输出：a * b = 30
        System.out.println("a / b = " + (a / b)); // 输出：a / b = 3 整数相除舍弃小数
        System.out.println("a % b = " + (a % b)); // 输出：a % b = 1 取余数

        // ===================== 019 一元算术运算符 ++自增 --自减 =====================
        int num1 = 5;
        num1++;
        System.out.println("后置自增num1++：" + num1); // 输出：后置自增num1++：6

        int num2 = 5;
        ++num2;
        System.out.println("前置自增++num2：" + num2); // 输出：前置自增++num2：6

        int num3 = 10;
        num3--;
        System.out.println("后置自减num3--：" + num3); // 输出：后置自减num3--：9

        // ===================== 020 赋值运算符 = += -= *= /= %= =====================
        int score = 60;
        score += 10;
        System.out.println("score += 10 → " + score); // 输出：score += 10 → 70
        score -= 5;
        System.out.println("score -= 5 → " + score); // 输出：score -= 5 → 65
        score *= 2;
        System.out.println("score *= 2 → " + score); // 输出：score *= 2 → 130

        // ===================== 021 关系运算符 > < >= <= == != 结果全是boolean =====================
        System.out.println("10 > 3 ：" + (a > b));  // 输出：10 > 3 ：true
        System.out.println("10 < 3 ：" + (a < b));  // 输出：10 < 3 ：false
        System.out.println("10 == 3：" + (a == b)); // 输出：10 == 3：false
        System.out.println("10 != 3：" + (a != b)); // 输出：10 != 3：true
        // System.out.println(a = b); // 易错：单个=是赋值，==才是判断相等

        // ===================== 022 逻辑运算符 &&短路与 ||短路或 !非 只操作布尔值 =====================
        boolean x = true;
        boolean y = false;
        System.out.println("true && false：" + (x && y)); // 输出：true && false：false 两边都真才真
        System.out.println("true || false：" + (x || y)); // 输出：true || false：true 一边真就真
        System.out.println("!true：" + (!x)); // 输出：!true：false 取反

        // 短路特性演示
        int m = 1;
        // false && 后面代码不会执行
        boolean res1 = (3 < 2) && (++m > 0);
        System.out.println("短路与后m值：" + m); // 输出：短路与后m值：1

        int n = 1;
        // true || 后面代码不会执行
        boolean res2 = (3 > 2) || (++n > 0);
        System.out.println("短路或后n值：" + n); // 输出：短路或后n值：1

        // ===================== 023 三元运算符 条件 ? 成立值 : 不成立值 =====================
        int age = 16;
        String result = age >= 18 ? "成年人" : "未成年人";
        System.out.println("三元判断年龄：" + result); // 输出：三元判断年龄：未成年人

        int max = a > b ? a : b;
        System.out.println("a和b最大值：" + max); // 输出：a和b最大值：10
    }
}
