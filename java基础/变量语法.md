public class B_AllDataType {
    public static void main(String[] args) {
        // ===================== 013 总纲：Java数据类型分为两大类 =====================
        // 1.基本数据类型：8种，值直接存在变量内
        // 2.引用数据类型：存内存地址，指向堆中真实数据（String、数组、自定义类）

        // ===================== 014 八大基本数据类型完整演示 =====================
        // 4种整数类型
        byte b = 10; // 占1字节，范围-128~127
        System.out.println("byte:" + b); // 输出：byte:10
        short s = 1000; // 占2字节
        System.out.println("short:" + s); // 输出：short:1000
        int i = 99999; // 默认整数类型，占4字节，最常用
        System.out.println("int:" + i); // 输出：int:99999
        long l = 9999999999L; // 长整型末尾必须加L标识，占8字节
        System.out.println("long:" + l); // 输出：long:9999999999

        // 2种浮点小数类型
        float f = 3.14F; // 单精度，末尾加F，占4字节
        System.out.println("float:" + f); // 输出：float:3.14
        double d = 3.1415926; // 默认小数类型，精度更高，占8字节
        System.out.println("double:" + d); // 输出：double:3.1415926

        // char字符：单引号包裹，仅单个字符，占2字节
        char c = 'A';
        System.out.println("char:" + c); // 输出：char:A

        // boolean布尔：只有true/false，用于判断
        boolean flag = true;
        System.out.println("布尔:" + flag); // 输出：布尔:true

        // ===================== 015 数据类型转换：自动转换 + 强制转换 =====================
        // 自动转换：小范围→大范围，系统自动执行
        byte numByte = 20;
        int numInt = numByte; // byte自动转int
        System.out.println("byte自动转int：" + numInt); // 输出：byte自动转int：20
        char ch = 'B';
        int chNum = ch; // char自动转int（B对应ASCII码66）
        System.out.println("char自动转int：" + chNum); // 输出：char自动转int：66

        // 强制转换：大范围→小范围，手动书写(目标类型)，会丢失精度
        double bigDec = 5.99;
        int smallInt = (int) bigDec; // 强制砍掉小数，不四舍五入
        System.out.println("double强制转int：" + smallInt); // 输出：double强制转int：5

        // 运算自动提升规则：不同类型运算统一转为更大范围类型
        byte m = 3;
        byte n = 4;
        // byte sumErr = m + n; // 报错：byte相加自动提升为int，不能直接赋值byte
        byte sum = (byte) (m + n);
        System.out.println("byte相加强转：" + sum); // 输出：byte相加强转：7

        // 不可转换规则
        // boolean bl = true;
        // int test = (int) bl; // 错误：boolean和所有数字类型不能互转

        // ===================== 016 引用数据类型（入门仅讲解String） =====================
        String str = "李四"; // String属于引用类型，双引号包裹一串文字
        System.out.println("引用类型字符串：" + str); // 输出：引用类型字符串：李四

        // 基础类型和字符串拼接会自动转为字符串
        String mix = "数字：" + 666;
        System.out.println("数字拼接字符串：" + mix); // 输出：数字拼接字符串：666
    }
}
