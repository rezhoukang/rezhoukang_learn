import java.util.concurrent.Executors;
import java.util.concurrent.ExecutorService;

public class J_ThreadAll {
    public static void main(String[] args) {
        // ===================== 125、126 进程与线程概念 =====================
        /*
        进程：电脑中独立运行软件（如微信、浏览器），独占内存资源
        线程：进程内独立执行分支，一个进程包含多条线程，线程共享进程资源
        多线程：同一时间执行多段代码，提升程序效率
        */
        // Thread中文线程
        // ===================== 127 创建线程（最佳实践：Runnable + Lambda） =====================
        // 推荐实现Runnable接口 + Lambda表达式，避免继承Thread的单继承限制，代码更简洁
        Thread t1 = new Thread(() -> {
            for (int i = 1; i <= 3; i++) {
                System.out.println("t1线程执行：" + i);
                try { Thread.sleep(500); } catch (InterruptedException e) { e.printStackTrace(); } // 打印异常堆栈，方便定位bug
            }
        });
        t1.start(); // 启动线程必须调start()，不能直接调run()

        // 主线程和t1并发执行，顺序不固定
        //下面这个是主线程
        for (int i = 1; i <= 3; i++) {
            System.out.println("主线程执行：" + i);
            try { Thread.sleep(500); } catch (InterruptedException e) { e.printStackTrace(); } // 打印异常堆栈
        }
        /* 输出特点：两条线程交替打印，顺序不固定，并发执行 */
        // 顺序不固定，但一种可能的结果是：
        // t1线程执行：1
        // 主线程执行：1
        // t1线程执行：2
        // 主线程执行：2
        // t1线程执行：3
        // 主线程执行：3

        // ===================== 128 线程生命周期5种状态 =====================
        /*
        1.新建：new线程对象
        2.就绪：调用start()，等待CPU分配时间片
        3.运行：CPU选中，执行run()
        4.阻塞：sleep/锁等待/wait，暂停执行
        5.终止：run执行完毕，线程销毁
        */


        // ===================== 129、130 sleep线程休眠 =====================
        // Thread.sleep(毫秒)：让当前线程暂停指定时间，释放CPU，不释放锁
        new Thread(() -> {
            System.out.println("休眠前");
            try {
                Thread.sleep(1000); // 休眠1秒
            } catch (InterruptedException e) {
                e.printStackTrace(); // 打印异常堆栈
            }
            System.out.println("休眠结束"); // 延迟1秒后输出
        }).start();


        // ===================== 132 线程池（提前招好工人，有活自动干） =====================
        ExecutorService pool = Executors.newFixedThreadPool(3); // 招3个工人
        pool.submit(() -> System.out.println("任务1")); // 来活了，工人自动干
        pool.submit(() -> System.out.println("任务2"));
        pool.submit(() -> System.out.println("任务3"));
        pool.submit(() -> System.out.println("任务4"));
        pool.submit(() -> System.out.println("任务5")); // 5个活，3个工人轮流干
        pool.shutdown(); // 活干完，辞退工人


        // ===================== 133~135 同步锁synchronized（上锁，防超卖） =====================
        // 两个窗口卖10张票，synchronized保证同一时间只有一个人操作
        class Ticket {
            int count = 10;
            void sell() {
                synchronized (this) { // 锁住，同时只允许一个人进
                    if (count <= 0) return; // 没票了
                    // Thread.currentThread() = 当前线程对象；getName() = 获取线程名字
                    System.out.println(Thread.currentThread().getName() + "卖第" + count + "张");
                    // 窗口1卖第10张
                    // 窗口2卖第9张
                    // 窗口1卖第8张
                    // 窗口2卖第7张
                    // 窗口1卖第6张
                    // 窗口2卖第5张
                    // 窗口1卖第4张
                    // 窗口2卖第3张
                    // 窗口1卖第2张
                    // 窗口2卖第1张
                    count--;
                }
            }
        }
        Ticket t = new Ticket();
        new Thread(() -> { for (int i=0; i<10; i++) t.sell(); }, "窗口1").start();
        new Thread(() -> { for (int i=0; i<10; i++) t.sell(); }, "窗口2").start();
        // 不加锁：两个窗口可能同时卖同一张票 → 超卖
        // 加锁后：窗口1卖完count--，窗口2才能进来，不会出负数
    }
}
