public class G_ExceptionDemo {
    public static void main(String[] args) {
        // ===================== 091、092 异常介绍 & 基础语法 try-catch-finally =====================
        /*
        异常：程序运行时出现的错误，不处理会直接终止程序；
        try：存放可能报错的代码；
        catch：捕获对应类型异常，写补救逻辑；
        finally：无论是否报错，最后一定会执行（常用于关闭资源）
        */
        try {
            // 模拟用户输入（实际开发中来自控制台/前端，无法提前预判）
            String userInput = "abc";
            int num = Integer.parseInt(userInput); // "abc"不能转数字，抛异常
        } catch (NumberFormatException e) {
            System.out.println("输入格式错误，请输入数字"); // 补救：提示用户
        } finally {
            System.out.println("finally块一定会执行"); // 资源清理（关闭文件、数据库连接等）
        }


        // ===================== 093、094、095 三类常见内置异常演示 =====================
        // 常用内置异常（API）：
        //   NullPointerException           调用了 null 对象的方法
        //   NumberFormatException          字符串转数字失败
        //   ClassCastException             强制类型转换失败
        //   ArrayIndexOutOfBoundsException 数组下标越界
        //   ArithmeticException            算术异常（如除零）
        // 1. NullPointerException 空指针异常
        try {
            String str = null;
            str.length(); // null调用方法，触发空指针
        } catch (NullPointerException e) {
            System.out.println("异常1：空指针，变量没有存储对象地址"); // 输出：异常1：空指针，变量没有存储对象地址
        }

        // 2. NumberFormatException 数字格式转换异常
        try {
            String numStr = "abc";
            Integer.parseInt(numStr); // 文字无法转为数字
        } catch (NumberFormatException e) {
            System.out.println("异常2：字符串不能转为数字"); // 输出：异常2：字符串不能转为数字
        }

        // 3. ClassCastException 类型转换异常
        try {
            Object obj = "文本";
            Integer num = (Integer) obj; // 字符串无法强转整数
        } catch (ClassCastException e) {
            System.out.println("异常3：类型强制转换失败"); // 输出：异常3：类型强制转换失败
        }


        // ===================== 096 异常抛出 throws + throw =====================
        // 执行顺序：行1→行2→跳到行6→行7→行8→回到行3→行4
        try {                                          // 行1：开始监视
            testThrow(0);                              // 行2：调用testThrow(0)，跳到行6
        } catch (ArithmeticException e) {              // 行3：异常从testThrow抛回这里
            System.out.println("收到：除数为零");         // 行4：catch接住异常
        }                                              // 行5：结束
    }                                                  // main结束

    // throws 声明该方法会抛出算术异常
    public static void testThrow(int num) throws ArithmeticException {
        if (num == 0) {                                // 行7：num=0 → true
            throw new ArithmeticException("发送：除数为零"); // 行8：抛异常，方法立刻结束，回到行3
        }
        int res = 10 / num;   // ❌ 不执行，上面已throw
        System.out.println("计算结果：" + res); // ❌ 不执行
    }
}

// 自定义异常继承Exception（编译期异常）
class AgeIllegalException extends Exception {
    // 构造方法，接收异常提示文字
    public AgeIllegalException(String msg) {
        super(msg);
    }
}

// 测试自定义异常
class CustomExceptionTest {
    public static void main(String[] args) {
        int age = -5;
        try {
            checkAge(age);
        } catch (AgeIllegalException e) {
            System.out.println("自定义异常提示：" + e.getMessage()); // 输出：自定义异常提示：年龄不能为负数
        }
    }

    // 校验年龄，不合法则抛出自定义异常
    public static void checkAge(int age) throws AgeIllegalException {
        if (age < 0 || age > 120) {
            throw new AgeIllegalException("年龄不能为负数");
        }
        System.out.println("年龄合法");
    }
}
